# What each completion script offers for a command line — the same in every
# shell (bash, zsh, fish, nu, PowerShell), checked by
# `tests/completions/rust/table_tests.rs`. One case per line:
#
#     <command line>|<candidates, in any order>[|<shells>]
#
# The word under the cursor is what follows the last space of the line (a
# line that ends with a space completes an empty word). A case that names
# shells replaces the general one of that line for them; a candidate that
# holds spaces is written in double quotes. Rusk never offers task ids; a
# word that is an id followed by a space gets the flags. The database the
# scripts read holds one task: `1 buy milk`.
rusk |a add c completions e edit g gen help l list m mark r restore rm s search serve sync
# fish and nu list the flags of rusk itself along with the commands.
rusk |-h --help -V --version a add c completions e edit g gen help l list m mark r restore rm s search serve sync|fish,nu
rusk -|-h --help -V --version
rusk --|--help --version
rusk a|a add
rusk s|s search serve sync
rusk r|r restore rm
rusk he|help
# Task ids in place of the command show those tasks; they are never
# offered, and after them comes --json, which prints the tasks as JSON
# lines. With it on the line, help only.
rusk 1|
rusk 1 |-h --help --json
rusk 1 2 |-h --help --json
rusk 1,2 -|-h --help --json
rusk 1 --j|--json
rusk 1 --json |-h --help
rusk 1 --json 2 -|-h --help
rusk --json |-h --help
rusk help |add completions edit gen list mark restore rm search serve sync
rusk help s|search serve sync
# `add --json` reads the tasks from standard input and goes alone: offered
# while nothing follows `add`; with it on the line, help only.
rusk add |-h --help --json
rusk add -|-h --help --json
rusk a |-h --help --json
rusk add --j|--json
rusk add --json |-h --help
rusk add --json -|-h --help
rusk add -d tomorrow |-h --help
rusk add x |-h --help -d --date -a --after
rusk add x -|-h --help -d --date -a --after
rusk add x --date |-h --help
rusk add x -d tomorrow |-h --help -a --after
rusk add x -a 1 -d tomorrow |-h --help
rusk edit |-h --help
rusk e |-h --help
rusk edit 1 |-h --help -d --date -a --after
# fish inserts one token per Tab: the id first, then the text on its own.
rusk edit 1 |"buy milk"|fish
rusk edit 1 -d tomorrow |-h --help -a --after
rusk edit 1 -|-h --help -d --date -a --after
rusk mark |-h --help -p --priority
rusk m |-h --help -p --priority
rusk mark 1 |-h --help -p --priority
# `rm` takes ids, never offered, or a word, `done` or `all`, offered
# until one of them or an id is on the line; `--older` and `--newer` go
# with a word and take an age or a date, a free value; `--yes` goes with all.
rusk rm |-h --help --older --newer -y --yes done all
rusk rm -|-h --help --older --newer -y --yes
rusk rm d|done
rusk rm a|all
rusk rm 1 |-h --help -y --yes
rusk rm 1 2 |-h --help -y --yes
rusk rm 1,2 -|-h --help -y --yes
rusk rm done |-h --help --older --newer -y --yes
rusk rm done --older |
rusk rm done --newer |
rusk rm done --older 7d |-h --help --older --newer -y --yes
rusk rm done --newer 1m --older |
rusk rm done --newer 01-09-2026 --older 30-09-2026 |-h --help --older --newer -y --yes
rusk rm all |-h --help --older --newer -y --yes
rusk rm all --older |
rusk rm all --newer 1y --older |
rusk rm all --older 1y |-h --help --older --newer -y --yes
rusk list |-h --help -c --compact --no-compact
rusk l |-h --help -c --compact --no-compact
rusk list -|-h --help -c --compact --no-compact
rusk search -|-h --help --id
rusk restore |-h --help
rusk r |-h --help
rusk gen -|-h --help -o --output
rusk serve -|-h --help --host --port
rusk sync |-h --help push pull
rusk completions |-h --help install show
rusk c |-h --help install show
rusk completions install |-h --help bash zsh fish nu powershell
rusk completions show |-h --help bash zsh fish nu powershell
rusk completions install bash |-h --help zsh fish nu powershell
# A command typed in full is a command: Tab ends the word, and the next Tab
# offers what follows it.
rusk add|add
rusk list|list
rusk e|e edit
rusk edit|edit
rusk rm|rm
rusk help|help
rusk sync|sync
rusk completions|completions
# `rusk help` takes a command, and a command of a command; nothing more.
rusk help add |
rusk help sync |push pull
rusk help completions |install show
rusk help sync push |
rusk help add x|
# The text of a task is not a command.
rusk add h|
rusk search h|
rusk add x h|
# A flag given with its value in one word is on the line too.
rusk add x --date=tomorrow |-h --help -a --after
rusk add x -dtomorrow |-h --help -a --after
rusk edit 1 --after=2 |-h --help -d --date
rusk edit 1 -a2 -d_ |-h --help
# After `--` every word is text.
rusk add x -- |
rusk edit 1 -- |
rusk add -- -|
# After a shell, the ones not on the line yet are offered, for `install` and
# `show` alike, until none is left. The word under the cursor is still being
# typed: a shell typed in full there is completed, not left out.
rusk completions show zsh |-h --help bash fish nu powershell
rusk completions show fish zsh |-h --help bash nu powershell
rusk completions install fish |-h --help bash zsh nu powershell
rusk completions install fish zsh |-h --help bash nu powershell
rusk completions install fish zsh nu |-h --help bash powershell
rusk completions install fish zsh nu bash |-h --help powershell
rusk completions install bash zsh fish nu powershell |-h --help
rusk completions install fish zsh p|powershell
rusk completions install fish zsh z|
rusk completions install fish|fish
rusk completions install fish zsh|zsh
rusk completions install fish f|
# fish offers the text of the task next to its id after `--` as well.
rusk edit 1 -- |"buy milk"|fish
